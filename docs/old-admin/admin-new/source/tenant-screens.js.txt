const field = ({
  id,
  label,
  value = "",
  placeholder = "",
  hint = "",
  mono = false,
  invalid = "",
  type = "text",
  copy = false,
}) => {
  const input =
    `<input id="${id}" type="${type}" class="form-control${mono ? " font-monospace" : ""}${invalid ? " is-invalid" : ""}"` +
    ` value="${value}"${placeholder ? ` placeholder="${placeholder}"` : ""}>`;
  return `
<div class="mb-3">
  <label class="form-label" for="${id}">${label}</label>
  ${copy ? `<div class="input-group"><div class="form-control font-monospace" style="height: auto; word-break: break-all">${value}</div><button type="button" class="btn">Copy</button></div>` : input}
  ${invalid && invalid !== true ? `<div class="invalid-feedback d-block">${invalid}</div>` : ""}
  ${hint ? `<div class="form-hint">${hint}</div>` : ""}
</div>`;
};
const tick = (label, description, checked = false, kind = "") => `
<label class="form-check${kind === "switch" ? " form-switch" : ""} mb-3">
  <input class="form-check-input" type="checkbox"${checked ? " checked" : ""}>
  <span class="form-check-label">${label}</span>
  <span class="form-check-description">${description}</span>
</label>`;

// A window, drawn where it stands rather than over a page.
const windowPart = ({ width = 540, status = "", title, content, foot }) => `

(async () => {
  const browser = await chromium.launch({
    executablePath: "/opt/pw-browsers/chromium",
  });
  for (const name of fs.readdirSync(OUT)) fs.rmSync(path.join(OUT, name));
  for (const [name, title, html, width] of screens) {
    const file = path.join(OUT, name + ".html");
    fs.writeFileSync(file, doc(title, html));
    const context = await browser.newContext({
      viewport: { width, height: 100 },
      deviceScaleFactor: 2,
    });
    const tab = await context.newPage();
    await tab.goto("file://" + file, { waitUntil: "load" });
    let height = await tab.evaluate(() =>
      Math.ceil(document.documentElement.scrollHeight),
    );
    await tab.setViewportSize({ width, height });
    height = await tab.evaluate(() =>
      Math.ceil(document.documentElement.scrollHeight),
    );
    await tab.setViewportSize({ width, height });
    await tab.waitForTimeout(200);
    await tab.screenshot({ path: path.join(OUT, name + ".png") });
    console.log(name, `${width}x${height}`);
    await context.close();
  }
  await browser.close();
})().catch((error) => {
  console.error(error);
  process.exit(1);
});
